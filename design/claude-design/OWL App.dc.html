export async function withRetry<T>(fn: () => Promise<T>, o: Opts) {
  for (let i = 0; ; i++) {
    try { return await fn(); }
    catch (e) { if (i >= o.tentativas || !retryable(e)) throw e; }
    await sleep(o.baseMs * o.fator ** i);
  }
}
